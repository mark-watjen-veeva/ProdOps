/**
 * Compiles unique checklist tracking dimensions and individual profiles from the database sheet.
 */
function fetchPodDataFromSheet() {
  try {
    let ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) {
      ss = SpreadsheetApp.openById(CONFIG.DEFAULT_SPREADSHEET_ID);
    }
    
    const sheet = ss.getSheetByName(CONFIG.PODS_SHEET);
    if (!sheet) {
      return { status: 'error', message: `Sheet labeled "${CONFIG.PODS_SHEET}" was not found.` };
    }

    const data = sheet.getDataRange().getDisplayValues();
    if (data.length <= 1) {
      return { status: 'success', individualPods: [], uniqueRegions: [], uniqueReleases: [], uniqueProducts: [] };
    }

    // Immutable hardcoded column indexes matching your spreadsheet layout explicitly
    const idxRelease = 0; // Column A ("Release")
    const idxPod = 2;     // Column C ("POD")
    const idxProduct = 3; // Column D ("Product")
    const idxRegion = 8;  // Column I ("Region")
    const idxUpgrade = 9; // Column J ("Upgrade Mode")

    let individualPods = [];
    let podMetadata = {};
    let regionsSet = new Set();
    let releasesSet = new Set();
    let productsSet = new Set();

    for (let i = 1; i < data.length; i++) {
      let row = data[i];
      
      let podName = String(row[idxPod]).trim();
      if (!podName) continue; 

      let relVal = String(row[idxRelease]).trim();
      let prodVal = String(row[idxProduct]).trim();
      let regVal = String(row[idxRegion]).trim();
      let upgVal = String(row[idxUpgrade]).trim();

      if (relVal.includes("/") || relVal.includes("-") || !isNaN(relVal.charAt(0))) {
        continue;
      }

      individualPods.push(podName);
      
      podMetadata[podName] = {
        region: regVal || "Unknown",
        release: relVal || "General",
        product: prodVal || "Vault",
        upgradeMode: upgVal || "Classic"
      };

      if (regVal && regVal.trim() !== "") regionsSet.add(regVal.trim());
      if (relVal && relVal.trim() !== "") releasesSet.add(relVal.trim());
      if (prodVal && prodVal.trim() !== "") productsSet.add(prodVal.trim());
    }

    individualPods.sort((a, b) => a.localeCompare(b, undefined, {numeric: true, sensitivity: 'base'}));

    return {
      status: 'success',
      individualPods: individualPods,
      podMetadata: podMetadata,
      uniqueRegions: Array.from(regionsSet).sort(),
      uniqueReleases: Array.from(releasesSet).sort(),
      uniqueProducts: Array.from(productsSet).sort()
    };

  } catch (err) {
    return { status: 'error', message: err.toString() };
  }
}

/**
 * Downloads layout parameters mapped inside the Templates workbook registry.
 */
function getSpreadsheetTemplatesConfig() {
  try {
    let ss = SpreadsheetApp.getActiveSpreadsheet();
    if (!ss) {
      ss = SpreadsheetApp.openById(CONFIG.DEFAULT_SPREADSHEET_ID);
    }
    
    const sheet = ss.getSheetByName(CONFIG.TEMPLATES_SHEET);
    if (!sheet) {
      return { status: 'error', message: `Configuration worksheet "${CONFIG.TEMPLATES_SHEET}" missing.` };
    }

    const data = sheet.getDataRange().getValues();
    if (data.length <= 1) {
      return { status: 'error', message: "The template configuration database is completely empty." };
    }

    // 🌟 FIXED: Immutable hardcoded coordinates mapping layout explicitly to bypass index search failures
    const idxKey = 0;      // Column A
    const idxLabel = 1;    // Column B
    const idxSubject = 2;  // Column C
    const idxTitle = 3;    // Column D
    const idxDowntime = 4; // Column E
    const idxUpper = 5;    // Column F
    const idxLower = 6;    // Column G
    const idxNotesUrl = 7; // Column H

    let templatesRegistry = [];

    for (let i = 1; i < data.length; i++) {
      let row = data[i];
      let key = String(row[idxKey]).trim();
      if (!key) continue;

      let rawDowntime = String(row[idxDowntime]).toLowerCase().trim();
      let isDowntimeTrue = (rawDowntime === "true" || rawDowntime === "yes" || rawDowntime === "1");

      templatesRegistry.push({
        key: key,
        label: String(row[idxLabel]).trim() || key,
        subject: String(row[idxSubject]).trim(),
        title: String(row[idxTitle]).trim(),
        requiresDowntime: isDowntimeTrue,
        htmlUpper: String(row[idxUpper]),
        htmlLower: String(row[idxLower]),
        releaseNotesLink: String(row[idxNotesUrl]).trim() || "https://vaulthelp.veeva.com"
      });
    }

    return { status: 'success', templates: templatesRegistry };

  } catch (e) {
    return { status: 'error', message: e.toString() };
  }
}

/**
 * Layout compilation core engine. Builds synchronized email layouts cleanly with CDN image banner assets.
 */
function generateMasterHtmlLayout(formData) {
  try {
    let rawVersion = formData.version || "26R1.0.22";
    let shortVersion = rawVersion.substring(0, 4);
    let cleanUrl = formData.notesUrl || "https://vaulthelp.veeva.com";

    let rawUpperText = formData.htmlUpper || "";
    let rawLowerText = formData.htmlLower || "";

    let cleanEnvTheme = formData.envTheme || "PROD";

    let topBannerImageUrl = cleanEnvTheme === "SBX"
      ? "http://cdn.mcauto-images-production.sendgrid.net/f3de798f1d5c4acf/4b9a6798-b8ad-475d-940b-67aaeb29039f/1578x93.png"
      : "http://cdn.mcauto-images-production.sendgrid.net/f3de798f1d5c4acf/39e112bf-4c92-4bcb-9fca-cf442dfbff2d/776x50.png";

    let bottomBannerImageUrl = cleanEnvTheme === "SBX"
      ? "http://cdn.mcauto-images-production.sendgrid.net/f3de798f1d5c4acf/f8bb1bde-21f8-4910-a6b4-ae373a014261/1578x50.png"
      : "http://cdn.mcauto-images-production.sendgrid.net/f3de798f1d5c4acf/66c81663-cafb-4ec2-851d-b6c10e25dfba/779x25.png";

    let topBannerMarkup = `<img src="${topBannerImageUrl}" alt="Top Banner" style="width: 100%; max-width: 600px; height: auto; display: block;" />`;
    let bottomBannerMarkup = `<img src="${bottomBannerImageUrl}" alt="Bottom Banner" style="width: 100%; max-width: 600px; height: auto; display: block;" />`;

    let processedHeaderHtml = String(rawUpperText)
      .replace(/\n/g, '<br/>')
      .replace(/\[\[VERSION\]\]/g, shortVersion)
      .replace(/\${cleanUrl}/g, cleanUrl);

    let tempHtml = String(rawLowerText).replace(/\n/g, '<br/>');

    let processedNotesHtml = tempHtml
      .replace(/Developer Release Notes/gi, "[[DEV_LINK_TOKEN]]");

    if (processedNotesHtml.includes("[[VERSION]] Release Notes")) {
      processedNotesHtml = processedNotesHtml.replace(/\[\[VERSION\]\] Release Notes/g, `<a href="${cleanUrl}" clicktracking="off" target="_blank" style="color: #007bff; text-decoration: none; font-weight: normal;">${shortVersion} Release Notes</a>`);
    } else {
      processedNotesHtml = processedNotesHtml.replace(/Release Notes/g, `<a href="${cleanUrl}" clicktracking="off" target="_blank" style="color: #007bff; text-decoration: none; font-weight: normal;">Release Notes</a>`);
    }

    processedNotesHtml = processedNotesHtml
      .replace(/\[\[DEV_LINK_TOKEN\]\]/g, "<a href='https://limited.veevavault.dev/rn/' clicktracking='off' target='_blank' style='color: #007bff; text-decoration: none;'>Developer Release Notes</a>");

    processedNotesHtml = processedNotesHtml
      .replace(/\[\[LOCATION_TAG\]\]/g, `<a href="https://rn.veevavault.help/en/gr/vault-pod-details/" clicktracking="off" target="_blank" style="color: #007bff; text-decoration: none;">here</a>`)
      .replace(/\[\[VERSION\]\]/g, shortVersion)
      .replace(/\${cleanUrl}/g, cleanUrl)
      .replace(/(Vault's POD location\.)(?:<br\s*\/?>\s*)+/i, '$1<hr style="border: 0; border-top: 2px solid #b1b1b1; margin-top: 15px; margin-bottom: 15px;"/>')
      .replace(/Veeva Trust Site/g, "<a href='https://trust.veeva.com' clicktracking='off' target='_blank' style='color: #007bff; text-decoration: none;'>Veeva Trust Site</a>")
      .replace(/Veeva Product Support/g, "<a href='https://support.veeva.com/hc/en-us' clicktracking='off' target='_blank' style='color: #007bff; text-decoration: none;'>Veeva Product Support</a>")
      .replace(/<hr\s*\/?>/gi, '<hr style="border: 0; border-top: 2px solid #b1b1b1; margin-top: 15px; margin-bottom: 15px;"/>');

    let downtimeModuleMarkup = "";
    if (formData.requiresDowntime === true) {
      downtimeModuleMarkup = `
        <tr>
          <td style="padding: 4px 0; font-family: Arial, sans-serif; font-size: 14px; color: #333333; line-height: 1.3;">
            <strong>Estimated Vault Downtime:</strong> ${formData.estimatedDowntime || '90 Minutes'}
          </td>
        </tr>
      `;
    }

    // Dynamic World Clock Anchor links
    let baseFriendlyDate = formData.friendlyDate || '';
    
    let cleanIso = "";
    if (formData.targetDateIso) {
      cleanIso = formData.targetDateIso;
    } else {
      let rawInputDate = formData.releaseDate || "2026-06-29"; 
      let rawInputTime = formData.releaseTime || "17:15";     
      cleanIso = rawInputDate.replace(/[-]/g, "") + "T" + rawInputTime.replace(/[:]/g, "");
    }

    let worldClockUrl = "https://www.timeanddate.com/worldclock/fixedtime.html?p1=5402&iso=" + cleanIso + "&sort=1";

    let rawPodsText = formData.affectedPods || "";
    let containsApac = /APAC/i.test(rawPodsText);
    let containsEu = /EU/i.test(rawPodsText);

    let friendlyDateParts = baseFriendlyDate.split(/\s*\/\s*/);
    let pstPart = friendlyDateParts[0] || baseFriendlyDate;
    let jstPart = "";
    let cestPart = "";

    if (friendlyDateParts.length > 1) {
      friendlyDateParts.forEach(part => {
        if (/JST/i.test(part)) jstPart = part;
        if (/CEST/i.test(part)) cestPart = part;
      });
    }

    if (!jstPart && containsApac) jstPart = "Tuesday June 30, 9:15 AM JST";
    if (!cestPart && containsEu) cestPart = "Tuesday June 30, 2:15 AM CEST";

    let timingLinkSegments = [];
    timingLinkSegments.push(`<a href="${worldClockUrl}" clicktracking="off" target="_blank" style="color: #007bff; text-decoration: none;">${pstPart}</a>`);
    
    if (containsEu && cestPart) {
      timingLinkSegments.push(`<a href="${worldClockUrl}" clicktracking="off" target="_blank" style="color: #007bff; text-decoration: none;">${cestPart}</a>`);
    }
    if (containsApac && jstPart) {
      timingLinkSegments.push(`<a href="${worldClockUrl}" clicktracking="off" target="_blank" style="color: #007bff; text-decoration: none;">${jstPart}</a>`);
    }

    let compiledWhenString = timingLinkSegments.join(" / ");

    // Automated Affected PODs Layout Formatter
    let affectedLabelLabels = [];
    affectedLabelLabels.push("US");
    if (containsEu) affectedLabelLabels.push("EU");
    if (containsApac) affectedLabelLabels.push("APAC");
    let affectedLabelText = affectedLabelLabels.join("/") + " PODs";

    let cleanAffectedPods = String(rawPodsText)
      .replace(/US General Release:/gi, 'US General Release:')
      .replace(/US Limited Release:/gi, '<br/><br/>US Limited Release:')
      .replace(/APAC General Release:/gi, '<br/><br/>APAC General Release:')
      .replace(/APAC Limited Release:/gi, '<br/><br/>APAC Limited Release:')
      .replace(/EU General Release:/gi, '<br/><br/>EU General Release:')
      .replace(/EU Limited Release:/gi, '<br/><br/>EU Limited Release:');

    let detailsTableMarkup = `
      <table border="0" cellpadding="0" cellspacing="0" width="100%" style="margin-bottom: 15px; border-collapse: collapse;">
        <tr>
          <td style="padding: 4px 0; font-family: Arial, sans-serif; font-size: 14px; color: #333333; line-height: 1.3;">
            <strong>What:</strong> Veeva Vault ${rawVersion}
          </td>
        </tr>
        <tr>
          <td style="padding: 4px 0; font-family: Arial, sans-serif; font-size: 14px; color: #333333; line-height: 1.3;">
            <strong>When:</strong> ${compiledWhenString}
          </td>
        </tr>
        ${downtimeModuleMarkup}
      </table>
    `;

    let podsBlockMarkup = `
      <div style="font-family: Arial, sans-serif; font-size: 14px; color: #333333; line-height: 1.5; margin-bottom: 20px; margin-top: 5px;">
        <strong>Affected PODs:</strong> ${affectedLabelText}<br/>
        ${cleanAffectedPods}
      </div>
    `;

    let bulletsBlockMarkup = `
      <div style="font-family: Arial, sans-serif; font-size: 14px; color: #333333; line-height: 1.5; margin-bottom: 20px; margin-top: 15px;">
        ${processedHeaderHtml}
      </div>
    `;

    // Comprehensive Case-Insensitive regular expression matching to trap upgrade configurations cleanly
    let targetTemplateString = String(formData.templateKey || "") + " " + String(formData.emailTitle || "");
    let isUpgradeTemplate = /classic|vbv|general|limited|release/i.test(targetTemplateString) && !/maintenance/i.test(targetTemplateString);

    // Dynamic horizontal rules asset tags
    let structuralDividerLine = `<hr style="border: 0; border-top: 2px solid #b1b1b1; margin-top: 15px; margin-bottom: 15px;"/>`;

    let coreBodyLayoutHTML = "";
    if (isUpgradeTemplate) {
      coreBodyLayoutHTML = `
        ${detailsTableMarkup}
        ${bulletsBlockMarkup}
        ${structuralDividerLine}
        ${podsBlockMarkup}
      `;
    } else {
      coreBodyLayoutHTML = `
        ${detailsTableMarkup}
        ${podsBlockMarkup}
        ${bulletsBlockMarkup}
        ${structuralDividerLine}
      `;
    }

    let fullMasterHtml = `
      <div style="background-color: #f4f7f9; padding: 25px; font-family: Arial, sans-serif;">
        <table align="center" border="0" cellpadding="0" cellspacing="0" width="100%" style="max-width: 600px; background-color: #ffffff; border-collapse: collapse; box-shadow: 0 2px 8px rgba(0,0,0,0.06); border-radius: 6px; overflow: hidden;">
          
          <tr>
            <td style="padding: 0; line-height: 0;">
              ${topBannerMarkup}
            </td>
          </tr>
          
          <tr>
            <td style="padding: 30px 25px 25px 25px; background-color: #ffffff;">
              <h2 style="font-family: Arial, sans-serif; font-size: 20px; font-weight: bold; color: #000000; margin-top: 0; margin-bottom: 18px; line-height: 1.2; text-align: center;">
                ${formData.emailTitle || 'Veeva Vault Release Notification'}
              </h2>

              ${coreBodyLayoutHTML}

              <div style="font-family: Arial, sans-serif; font-size: 14px; color: #333333; line-height: 1.5; margin-top: 15px;">
                ${processedNotesHtml}
              </div>
            </td>
          </tr>
          
          <tr>
            <td style="padding: 0; line-height: 0;">
              ${bottomBannerMarkup}
              <div style="padding: 15px 25px; background-color: #f9f9f9; font-family: Arial, sans-serif; font-size: 12px; color: #666666; line-height: 1.5; border-top: 1px solid #eeeeee;">
                You are receiving this email as you are subscribed to receive product announcement emails on your Vault user profile.
              </div>
            </td>
          </tr>
          
        </table>
      </div>
    `;

    return fullMasterHtml;

  } catch (err) {
    return `<div style="color:red; padding:20px;">Error generating preview template engine output: ${err.toString()}</div>`;
  }
}
